# 宝石模式（gem mode）实施计划

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 给 RoK 助手加一个「宝石模式」——以模拟器为单位，长按地图按钮进资源查看视角，回字形螺旋扫图 + YOLO 找宝石，点开放大确认后派兵采集，直到右侧队列满，等 20~30 分钟再下一轮。

**Architecture:** 复用现有多模式框架（`AppMode.GEM` 已在 `infra/config.py` 里、只是被 `IMPLEMENTED_MODES` 锁着）与 `WorkerRunner`。新增三块：`infra/config.py` 的 `GemConfig` 配置段、`workers/gem_sm.py` 状态机、`coordination/base_runtime.py` + `coordination/gem_runtime.py` 协调器。**不引入** EventBus 跨实例通信、`ActionLedger`、`RallyEventTracker`、集结前置门槛——宝石模式一台机器一个号、各干各的，这些寨子专属件一个都用不上（spec §2）。

**Tech Stack:** Python 3.11 / pydantic v2 / PyQt6 / OpenCV / ONNX Runtime / YOLOv8 / pytest。

**Spec:** `docs/superpowers/specs/2026-10-17-宝石模式-design.md`（本计划逐节实现它；执行者两篇都要读）

## Global Constraints

- **一律用 `.venv/Scripts/python.exe`**，下面记作 `$PY`。别用系统 `python`（那是 torch+cpu，会静默降级跑 CPU）。
- **测试默认只跑相关子集**；收尾才跑 `$PY -m pytest tests/ -q`（全套 ~5.5 分钟，2026-10-16 基线 1098 passed）。
- **任何失败都必须写成「干净中止本轮」**：设 `ctx["fail_reason"]` + 走到 `END` 边。**绝不抛异常**——`WorkerRunner` 收到 step 异常不重建 SM，会在同一状态退避重放，攒满 6 次直接停机（PROGRESS 已知问题 8 / 14，已经烧过两次）。
- **bash 双引号里不要出现反引号**（会被当命令替换执行）；写文件用单引号定界符的 heredoc。
- **删任何图片 / 模板 / 标注前，先确认没有引用**（用户明确要求：千万不要把在用的清理掉）。

```python
# _probe_gem_capture.py
"""宝石模式真机抓帧器（Stage 0 探针，验收后删）。

每步落一帧到 recordings/gem_probe/：
  01_map            地图视图基线（队列栏里**要有**在外的队列）
  02_longpress      长按 map_btn(95,990) 800ms 后的环形菜单
  03_resource_view  点环形菜单「资源」按钮之后
  04..07            四个方向各拖一次，用来量「一次 swipe 移动多少像素」

用法（仓库根目录）：
    .venv/Scripts/python.exe -X utf8 _probe_gem_capture.py
"""
from __future__ import annotations

import sys
import time
from pathlib import Path

import cv2

from rok_assistant.core.handle_source import create_handle_source
from rok_assistant.infra.app_paths import user_dir
from rok_assistant.infra.config import AppMode, load_config

OUT = user_dir() / "recordings" / "gem_probe"
MAP_BTN = (95, 990)            # map_btn ROI [0,900,250,1080] 的中心
RADIAL_RESOURCE_BTN = (428, 822)   # spec §3.1，取自 142653
DRAG_CENTER = (960, 540)
DRAG_PX = 800


def shot(handle, name: str) -> None:
    OUT.mkdir(parents=True, exist_ok=True)
    frame = handle.capture()
    # 不用 cv2.imwrite：非 ASCII 路径下它静默返回 False（同 runner.py 的坑）
    ok, buf = cv2.imencode(".png", frame)
    if not ok:
        raise RuntimeError(f"编码失败：{name}")
    (OUT / f"{name}.png").write_bytes(buf.tobytes())
    print(f"[shot] {name}: {frame.shape}", flush=True)


def main() -> int:
    cfg = load_config(Path("config.yaml"))
    inst = cfg.instances_for(AppMode.ZHAIDUO)[0]
    handle = create_handle_source(
        mumu_index=inst.mumu_index,
        mumu_manager_path=cfg.app.mumu_manager_path,
        adb_address=inst.adb_address,
        adb_path=cfg.app.adb_path,
        window_title_pattern=inst.window_title_pattern)

    shot(handle, "01_map")
    handle.swipe(*MAP_BTN, *MAP_BTN, 800)      # 长按 800ms
    time.sleep(1.5)
    shot(handle, "02_longpress")
    handle.click(*RADIAL_RESOURCE_BTN)
    time.sleep(2.0)
    shot(handle, "03_resource_view")

    for name, (dx, dy) in (("04_drag_right", (DRAG_PX, 0)),
                           ("05_drag_left", (-DRAG_PX, 0)),
                           ("06_drag_down", (0, DRAG_PX)),
                           ("07_drag_up", (0, -DRAG_PX))):
        x, y = DRAG_CENTER
        handle.swipe(x, y, x + dx, y + dy, 500)
        time.sleep(1.5)
        shot(handle, name)
    print("完成。下一步跑 _probe_gem_metrics.py。", flush=True)
    return 0


if __name__ == "__main__":
    sys.exit(main())
```

```python
# _probe_gem_metrics.py
"""宝石探针帧的数值判据打印器（Stage 0）。

**刻意不把图读进上下文**：截图 1920x1080 按面积计价，读一张比跑十次本脚本都贵。
这里只打印数字，人看数字下结论。

用法：
    .venv/Scripts/python.exe -X utf8 _probe_gem_metrics.py > recordings/gem_probe/metrics.txt
"""
from __future__ import annotations

import sys
from pathlib import Path

import cv2
import numpy as np
import yaml

FRAMES = Path("recordings/gem_probe")
MANIFEST = Path("templates/manifest.yaml")
QUEUE_ROI = (1700, 150, 1920, 850)     # 与 manifest 的 queue_* 同一块
MINIMAP_ROI = (1620, 10, 1880, 210)    # 验证它是否压住 queue_badge
PROBE_IDS = ("search_icon", "alliance_btn", "map_btn", "queue_panel",
             "queue_badge", "queue_gather_icon", "queue_march_icon",
             "queue_flag_icon", "queue_battle_icon", "queue_return_icon",
             "join_create_btn", "form_title", "march_btn")


def imread_unicode(path: Path) -> np.ndarray:
    return cv2.imdecode(np.fromfile(str(path), dtype=np.uint8), cv2.IMREAD_COLOR)


def edge_density(img: np.ndarray) -> float:
    """Canny 边缘像素占比。资源查看视角把地图缩成小图标 → 边缘密度显著上升。"""
    gray = cv2.cvtColor(img, cv2.COLOR_BGR2GRAY)
    return float(cv2.Canny(gray, 60, 160).mean() / 255.0)


def crop(img, roi):
    x1, y1, x2, y2 = roi
    return img[y1:y2, x1:x2]


def main() -> int:
    raw = yaml.safe_load(MANIFEST.read_text(encoding="utf-8"))
    entries = {e["id"]: e for e in raw.get("templates", [])}

    for path in sorted(FRAMES.glob("*.png")):
        img = imread_unicode(path)
        print(f"\n=== {path.name} {img.shape} ===")
        print(f"  整帧边缘密度   : {edge_density(img):.4f}")
        print(f"  队列栏边缘密度 : {edge_density(crop(img, QUEUE_ROI)):.4f}")
        print(f"  小地图边缘密度 : {edge_density(crop(img, MINIMAP_ROI)):.4f}")

        for tid in PROBE_IDS:
            e = entries.get(tid)
            if e is None:
                continue
            tpath = MANIFEST.parent / e["file"]
            if not tpath.is_file():
                print(f"  {tid:22s}: 模板文件缺失 {tpath.name}")
                continue
            tmpl = imread_unicode(tpath)
            roi = e.get("roi")
            search = img if roi in (None, "full") else crop(img, tuple(roi))
            if (search.shape[0] < tmpl.shape[0]
                    or search.shape[1] < tmpl.shape[1]):
                print(f"  {tid:22s}: 跳过（ROI 比模板小）")
                continue
            res = cv2.matchTemplate(search, tmpl, cv2.TM_CCOEFF_NORMED)
            print(f"  {tid:22s}: max={res.max():.3f}  "
                  f">=0.9 命中像素={int((res >= 0.9).sum())}")

        # 队列栏横带 std：队列真在外时这些带里会有图标 → std 明显高于空槽
        col = crop(img, QUEUE_ROI)
        bands = [col[i:i + 70] for i in range(0, col.shape[0] - 70, 70)]
        print("  队列栏横带 std : "
              + " ".join(f"{float(b.reshape(-1, 3).std(axis=0).mean()):.1f}"
                         for b in bands))
    return 0


if __name__ == "__main__":
    sys.exit(main())
```

```python
class InstanceBase(BaseModel):
    """一个模拟器实例的公共字段与校验。

    抽出来是因为宝石模式也要「一台模拟器一个账号」，字段与校验一字不差；
    复制一份的下场是两份 `_instance_checks` 慢慢漂移（PROGRESS 已知问题 16
    就是同一类漂移）。**只放两模式共有的东西**：mumu/adb 二选一。
    """

    id: str
    name: str = ""
    mumu_index: int | None = Field(default=None, ge=0)
    adb_address: str = ""
    window_title_pattern: str = ""  # Win32 备用路线；ADB 路线用不到

    @model_validator(mode="after")
    def _instance_checks(self):
        if self.mumu_index is None and not self.adb_address:
            raise ValueError(f"Instance {self.id}: mumu_index 或 adb_address 必须填一个")
        if self.mumu_index is not None and self.adb_address:
            raise ValueError(f"Instance {self.id}: mumu_index 与 adb_address 只能填一个")
        return self


class InstanceConfig(InstanceBase):
    characters: list[CharacterConfig]

    @model_validator(mode="after")
    def _character_names_unique(self):
        # leader/either 要求在 RootConfig 层校验：member-only 实例可跨实例 fill leader
        names = [c.name for c in self.characters]
        if len(names) != len(set(names)):
            raise ValueError(f"Instance {self.id} has duplicate character names")
        return self
```

```python
    字段取值只在这里校验上下界；**具体配几级 / 等几分钟不写进文档**，

    scan_drag_px: int = Field(default=800, ge=100)

    idle_wait_max: float = Field(default=30.0, gt=0)
```

```python
# _probe_gem_locate.py
"""在 imgs/gem 的帧里定位宝石/采集框候选，**只打印数字**。

宝石图标是白色六边形晶体（142653 的 (1010,450) 附近），与白色的木材/粮食
嫩芽、铁矿螺旋形状不同。这里按「亮度 + 形状紧凑度」把候选列出来，人再挑。

用法：
    .venv/Scripts/python.exe -X utf8 _probe_gem_locate.py
"""
from __future__ import annotations

import sys
from pathlib import Path

import cv2
import numpy as np

SRC = Path("imgs/gem")
# 地图主体：避开顶部 HUD、底部按钮栏、右侧队列栏
MAP_ROI = (0, 90, 1620, 900)


def imread_unicode(p: Path) -> np.ndarray:
    return cv2.imdecode(np.fromfile(str(p), dtype=np.uint8), cv2.IMREAD_COLOR)


def white_blobs(img: np.ndarray):
    """白色小连通块：亮度高、面积适中。"""
    x1, y1, x2, y2 = MAP_ROI
    roi = img[y1:y2, x1:x2]
    gray = cv2.cvtColor(roi, cv2.COLOR_BGR2GRAY)
    mask = (gray > 200).astype(np.uint8) * 255
    n, labels, stats, cents = cv2.connectedComponentsWithStats(mask, 8)
    out = []
    for i in range(1, n):
        x, y, w, h, area = stats[i]
        if not (200 <= area <= 4000):
            continue
        if not (0.5 <= w / max(1, h) <= 2.0):
            continue
        out.append((int(cents[i][0] + x1), int(cents[i][1] + y1),
                    int(w), int(h), int(area)))
    return sorted(out, key=lambda t: -t[4])


def red_blobs(img: np.ndarray):
    """偏红的块：放大后的宝石矿（142730）。"""
    x1, y1, x2, y2 = MAP_ROI
    roi = img[y1:y2, x1:x2].astype(int)
    b, g, r = roi[..., 0], roi[..., 1], roi[..., 2]
    mask = ((r - b >= 60) & (r >= 140)).astype(np.uint8) * 255
    n, labels, stats, cents = cv2.connectedComponentsWithStats(mask, 8)
    out = []
    for i in range(1, n):
        x, y, w, h, area = stats[i]
        if area < 400:
            continue
        out.append((int(cents[i][0] + x1), int(cents[i][1] + y1),
                    int(w), int(h), int(area)))
    return sorted(out, key=lambda t: -t[4])


def main() -> int:
    for path in sorted(SRC.glob("*.png")):
        img = imread_unicode(path)
        print(f"\n=== {path.name} {img.shape} ===")
        w = white_blobs(img)
        print(f"  白色候选 {len(w)} 个，面积前 8：")
        for cx, cy, bw, bh, area in w[:8]:
            print(f"    ({cx},{cy}) {bw}x{bh} area={area}")
        r = red_blobs(img)
        if r:
            print(f"  红色候选 {len(r)} 个，面积前 5：")
            for cx, cy, bw, bh, area in r[:5]:
                print(f"    ({cx},{cy}) {bw}x{bh} area={area}")
    return 0


if __name__ == "__main__":
    sys.exit(main())
```

```python
# _probe_gem_threshold.py
"""对 imgs/gem 的 11 帧跑一遍 7 条宝石模板，打印每条的 max 分。

要看的不是「命中多少」，是**正样本帧与负样本帧的分差**：
  - gem_icon 应在 142653/142658/142707/142720/142725 上高分，在 142743/142750
    （已经在采集框里，地图被面板盖住）上低分。
  - collect_dialog / collect_btn 反过来。

用法：
    .venv/Scripts/python.exe -X utf8 _probe_gem_threshold.py
"""
from __future__ import annotations

import sys
from pathlib import Path

import cv2
import numpy as np
import yaml

SRC = Path("imgs/gem")
MANIFEST = Path("templates/manifest.yaml")
IDS = ("gem_icon", "gem_mine_zoom", "view_radial_menu", "view_btn_resource",
       "collect_dialog", "collect_btn", "back_from_detail")


def imread_unicode(p: Path) -> np.ndarray:
    return cv2.imdecode(np.fromfile(str(p), dtype=np.uint8), cv2.IMREAD_COLOR)


def main() -> int:
    raw = yaml.safe_load(MANIFEST.read_text(encoding="utf-8"))
    by_id = {e["id"]: e for e in raw["templates"]}
    frames = sorted(SRC.glob("*.png"))
    header = "模板".ljust(20) + "".join(f.name[5:17].rjust(14) for f in frames)
    print(header)
    for tid in IDS:
        e = by_id.get(tid)
        if e is None:
            print(f"{tid:20s} manifest 里没有")
            continue
        tmpl = imread_unicode(MANIFEST.parent / e["file"])
        row = f"{tid:20s}"
        for f in frames:
            img = imread_unicode(f)
            roi = e.get("roi")
            if roi in (None, "full"):
                search = img
            else:
                x1, y1, x2, y2 = roi
                search = img[y1:y2, x1:x2]
            if (search.shape[0] < tmpl.shape[0]
                    or search.shape[1] < tmpl.shape[1]):
                row += f"{'--':>14}"
                continue
            res = cv2.matchTemplate(search, tmpl, cv2.TM_CCOEFF_NORMED)
            row += f"{res.max():>14.3f}"
        print(row)
    return 0


if __name__ == "__main__":
    sys.exit(main())
```

```python
class GemStateMachine(StateMachine):
    """宝石模式：扫图找宝石 → 确认 → 派兵采集 → 队列满了就等。"""

    def __init__(self, handle_source, recognizers, *, queue_limit=5,
                 scan_rings=3, scan_drag_px=800, scan_drag_ms=500,
                 idle_wait_min=20.0, idle_wait_max=30.0, char_id="?",
                 human=None, rng=None, clock=time.time, sleeper=time.sleep):
        self._handle = handle_source
        self._rec = recognizers
        self._queue_limit = int(queue_limit)
        self._scan_rings = int(scan_rings)
        self._scan_drag_px = int(scan_drag_px)
        self._scan_drag_ms = int(scan_drag_ms)
        self._idle_wait_min = float(idle_wait_min) * 60.0
        self._idle_wait_max = float(idle_wait_max) * 60.0
        # char_id 只用于日志前缀：一台模拟器一个号，队列/宝石都是本机的
        # 事实，不需要跨实例归属（寨子模式的账本才需要按 char_id 归属）
        self._char_id = char_id
        # clock / sleeper 可注入：单测换成虚拟时钟，不烧真实的 20~30 分钟
        self._clock = clock
        self._sleep = sleeper
        self._rng = rng if rng is not None else random.Random()
        self._wait_deadline = 0.0
        self._scan_index = 0
        self._tried_gems: set[tuple[int, int]] = set()
        self._gem_fail_streak = 0
        self._pending_gem: tuple[float, float] | None = None
        super().__init__(initial="IDLE", human=human)
        self.last_image = None

    # ---- 转移表 ----

    def _setup(self):
        self.add_transition("IDLE", "NORMALIZE", self._normalize_view)
        self.add_transition("NORMALIZE", "END", self._abort_round,
                            guard=lambda ctx: ctx.get("fail_reason")
                            == NORMALIZE_STUCK_REASON)
        self.add_transition("NORMALIZE", "COUNT_QUEUES", self._count_queues)

    # ---- step 覆写：WAIT 不阻塞线程 ----

    def step(self, context: dict | None = None) -> None:
        """WAIT 状态**不阻塞线程**：每拍只看一眼 deadline，没到就原地空转。

        在 action 里 sleep(20~30 分钟) 也能「实现等待」，但那样
        `WorkerRunner.stop()` 要等这个 sleep 跑完才 join 得上——用户点停止
        会卡住最长半小时（`_STOP_TIMEOUT` 只有 2s）。改成轮询 deadline，
        与 either_sm 的 WAIT_RETURN 同一套路。
        """
        if self.current == "WAIT" and self._clock() < self._wait_deadline:
            return
        super().step(context)

    def is_terminal(self) -> bool:
        return self.current == "END"

    # ---- IDLE → NORMALIZE ----

    def _normalize_view(self, ctx):
        """把画面收拾到「地图视图」。

        与寨子模式不同：宝石模式**只认地图视图**（`alliance_btn` 可见）。
        城市视图、战争列表、搜索面板、残留表单都收拾掉；收拾不掉就写
        `normalize_stuck` 干净中止本轮（**不抛异常**）。
        """
        for _ in range(_NORMALIZE_ATTEMPTS):
            if self._find("net_error_confirm") is not None:
                self._handle.click(1638, 120)       # 断网弹框的确认
                continue
            if self._find("alliance_btn") is not None:
                ctx.pop("fail_reason", None)
                return
            if self._find("form_title") is not None:
                self._handle.click(1716, 69)        # 创建部队表单的 X
                continue
            if self._find("war_title") is not None:
                self._handle.click(1671, 64)        # 战争列表的 X
                continue
            if self._find("queue_panel") is not None:
                self._handle.click(1550, 320)       # 点空地收起队列侧栏
                continue
            if self._find("map_btn") is not None:
                self._click("map_btn")              # 城市视图 → 回地图
                continue
            self._sleep(self._pause(None))
        ctx["fail_reason"] = NORMALIZE_STUCK_REASON
        logger.warning("[宝石] 归一化收拾不出地图视图，本轮中止")

    # ---- 收尾出口 ----

    def _abort_round(self, ctx):
        """干净中止本轮：fail_reason 已由上游写好，这里只留日志。"""
        logger.warning("[宝石 %s] 本轮中止：%s", self._char_id,
                       ctx.get("fail_reason"))

    def _end_round(self, ctx):
        """本轮正常结束（**不是失败**）。

        两种来源：螺旋扫空，或队列满了且本轮已经等过一次。都不写
        fail_reason —— runner 据此记 rounds_ok；写了会被记成失败。
        """
        logger.info("[宝石 %s] 本轮结束（队列 %s/%s，扫了 %s 格）",
                    self._char_id, ctx.get("queue_count"), self._queue_limit,
                    self._scan_index)

    # ---- 占位：后续 task 逐个替换 ----

    def _count_queues(self, ctx):
        raise NotImplementedError
```

```python
    def _count_queues(self, ctx):
        """数右侧队列栏里在外的队列图标（主判据，spec §6.2）。

        已知局限（spec §12）：数的是**不同图标类型的命中数**，所以同一类型

        同类图标，这条局限就不存在；若证明是分行显示，改成按行计数。

        if self._find("net_error_confirm") is not None:
            self._handle.click(1638, 120)       # 断网弹框盖住队列栏
            ctx["queue_count"] = -1
            ctx["queues_full"] = False
            return
        count = self._count_queue_icons()
        if count < 0:
            ctx["fail_reason"] = QUEUE_UNKNOWN_REASON
            ctx["queues_full"] = False
            logger.warning("[宝石] 没有可用的队列图标判据（manifest 缺 "
                           "queue_*_icon），本轮中止")
            return
        ctx["queue_count"] = count
        ctx["queues_full"] = count >= self._queue_limit
        # 徽标只在日志里做交叉验证，**不参与裁决**（spec §6.2）：小地图与它
        # 的 ROI 重叠，世界地图上读出来的 N/M 不可信
        badge = self._find("queue_badge")

                    self._queue_limit, ctx["queues_full"],
                    "在场" if badge is not None else "不可见")

    def _count_queue_icons(self) -> int:
        """一帧里数命中的队列图标识别器个数。**一个识别器都不在 = -1**。

        -1 与 0 必须分开：0 是「数出来队列空着」（可以继续派兵），-1 是
        「根本没有判据」（manifest 没配、或装配失败）。

        self.last_image = img
        return sum(1 for rid in available
                   if self._rec[rid].recognize(img).matched)
```

```python
    # ---- COUNT_QUEUES → OPEN_RESOURCE_VIEW ----

    def _open_resource_view(self, ctx):
        """长按地图按钮唤起环形菜单 → 点「资源」→ 确认进了资源查看视角。

        判据是**相对边缘密度**（spec §8.2.1）：进视角后城堡/资源/野怪都缩成
        小图标，同样面积里 Canny 边缘像素显著变多。取比值而不是绝对阈值——
        绝对阈值会被地形（山/海）和昼夜光照带跑。
        """
        before = self._edge_density()
        self._long_press(_MAP_BTN)
        if not self._wait_for("view_radial_menu", _VIEW_TIMEOUT):
            ctx["fail_reason"] = RADIAL_MENU_MISSING_REASON
            logger.warning("[宝石] 长按 %s 未唤出环形菜单（菜单没出来或点偏了）",
                           _MAP_BTN)
            return
        if not self._click("view_btn_resource"):
            ctx["fail_reason"] = RADIAL_MENU_MISSING_REASON
            logger.warning("[宝石] 环形菜单在场但「资源」按钮没认出来")
            return
        self._sleep(_RADIAL_SETTLE)
        after = self._edge_density()
        if after < before * _VIEW_EDGE_RATIO:
            ctx["fail_reason"] = RESOURCE_VIEW_FAILED_REASON
            logger.warning("[宝石] 点「资源」后边缘密度 %.4f → %.4f（未达 %.1f 倍），"
                           "判为没进资源查看视角", before, after, _VIEW_EDGE_RATIO)
            return
        logger.info("[宝石] 已进资源查看视角（边缘密度 %.4f → %.4f）", before, after)
        self._reset_scan(ctx)

    def _reset_scan(self, ctx):
        """螺旋归零。**`_tried_gems` 每轮清一次**：上一轮确认失败的那些点位
        在下一轮是新的（队列排空后资源会刷新），不清会越扫越早收工。"""
        self._scan_index = 0
        self._tried_gems.clear()
        self._gem_fail_streak = 0
        self._pending_gem = None
        ctx.pop("fail_reason", None)
        ctx["scan_verdict"] = "continue"

    def _long_press(self, xy, ms: int = _LONG_PRESS_MS):
        """原地 swipe 就是长按（MockHandleSource / Adb / Win32 三个实现都支持）。"""
        x, y = xy
        self._handle.swipe(int(x), int(y), int(x), int(y), int(ms))

    def _edge_density(self) -> float:
        """地图主体 ROI 的 Canny 边缘像素占比（spec §8.2.1）。

        ROI 刻意避开顶部 HUD、底部按钮栏、右侧队列栏——这三块是常驻的，
        算进去会把「进没进视角」这个信号淹掉。
        """
        img = self._handle.capture()
        self.last_image = img
        x1, y1, x2, y2 = _VIEW_ROI
        gray = cv2.cvtColor(img[y1:y2, x1:x2], cv2.COLOR_BGR2GRAY)
        return float(cv2.Canny(gray, 60, 160).mean() / 255.0)
```

```python
# 螺旋每一步的位移方向（屏幕坐标，y 向下）。段长 1,1,2,2,3,3…，方向按
# 右→下→左→上 循环——展开就是一圈圈扩大的方框（用户说的「回字形」）。
_SPIRAL_DIRS = ((1, 0), (0, 1), (-1, 0), (0, -1))


def _scan_total_steps(rings: int) -> int:
    """N 圈的螺旋总步数：段长 1,1,2,2,…,N,N 之和 = N(N+1)。"""
    return max(0, int(rings)) * (max(0, int(rings)) + 1)


def _scan_vector(k: int) -> tuple[int, int]:
    """螺旋第 k 步（k 从 1 起）的方向，单位是「格」。

    直接循环求段号。闭式公式在段号奇偶上极易写错（本函数第一版就错在
    k=5 落在了「左」而不是「上」），而段数最多 scan_rings 段，循环成本
    可以忽略——**这里选好读好验的那个**，`test_scan_vector_follows_square_spiral`
    是它的哨兵。
    """
    leg, remaining = 0, int(k)
    while True:
        length = leg // 2 + 1
        if remaining <= length:
            break
        remaining -= length
        leg += 1
    return _SPIRAL_DIRS[leg % 4]
```

```python
    # ---- SCAN：回字形螺旋 ----

    _TRIED_GRID = 40        # 「同一颗宝石」的判定格：40px 内的命中算同一颗

    def _scan_step(self, ctx):
        """先看**当前**画面有没有宝石，没有就往外拖一格。

        顺序很关键：`_scan_index` 是「已经拖了几格」，所以每次进来先判断
        眼前这一屏，再决定拖不拖——否则第一屏会被整屏跳过。
        """
        ctx.pop("fail_reason", None)
        img = self._handle.capture()
        self.last_image = img
        hit = self._find_gem(img)
        if hit is not None and not self._is_tried(hit):
            self._pending_gem = hit
            ctx["scan_verdict"] = "gem"
            logger.info("[宝石] 第 %s 格发现候选 (%.0f,%.0f)",
                        self._scan_index, hit[0], hit[1])
            return
        if hit is not None:
            logger.info("[宝石] 第 %s 格的候选 (%.0f,%.0f) 之前确认失败过，跳过",
                        self._scan_index, hit[0], hit[1])
        if self._scan_index >= _scan_total_steps(self._scan_rings):
            ctx["scan_verdict"] = "exhausted"
            logger.info("[宝石] 螺旋扫完 %s 圈（%s 格）无新宝石",
                        self._scan_rings, self._scan_index)
            return
        self._scan_index += 1
        dx, dy = _scan_vector(self._scan_index)
        self._drag(dx * self._scan_drag_px, dy * self._scan_drag_px)
        self._sleep(_SCAN_SETTLE)
        ctx["scan_verdict"] = "continue"

    def _drag(self, dx: int, dy: int):
        """把画面往 (dx, dy) 方向挪：**手指反向拖**（手往左拖 = 看到右边）。

        起点固定在地图中心偏左：落在 HUD 或右侧队列栏上会变成「点击」，
        而点中地图上的资源会弹出详情面板，把整轮带偏。
        """
        x, y = _DRAG_CENTER
        self._handle.swipe(x, y, x - dx, y - dy, self._scan_drag_ms)

    def _find_gem(self, img):
        """在**已截好的一帧**里找宝石图标。返回中心坐标或 None。

        `gem_icon` 在 manifest 里是 `template_match` + YOLO 模型 → 装配成
        `RecognizerChain`（YOLO 腿在前），这里不需要知道是哪条腿命中的。
        """
        rec = self._rec.get("gem_icon")
        if rec is None:
            return None
        r = rec.recognize(img)
        return r.bbox.center() if r.matched else None

    def _gem_key(self, xy) -> tuple[int, int]:
        """把坐标量化成格：`gem_icon` 的检测框会随帧抖动几像素，不比格
        的话同一颗宝石会被当成新候选反复点开。"""
        return (int(xy[0]) // self._TRIED_GRID, int(xy[1]) // self._TRIED_GRID)

    def _is_tried(self, xy) -> bool:
        return self._gem_key(xy) in self._tried_gems
```

```python
    # ---- SCAN → ZOOM_CONFIRM ----

    def _zoom_confirm(self, ctx):
        """点开候选宝石 → 放大 → 认出「放大的宝石图标」才算数。

        双确认的第一腿（spec §6.6）：扫描时的 `gem_icon` 是小图标，误检率
        天然高于放大后的 `gem_mine_zoom`——放大这一步把「地图上一小坨白色」
        变成「一个明确的宝石矿面板」，是**成本最低的消歧手段**（一次点击
        + 一次截图）。第二腿是 COLLECT_DIALOG 的 OCR 文字确认。
        """
        ctx.pop("fail_reason", None)
        if not self._rec.get("gem_dialog_title"):
            # 缺腿的确认策略不静默降级——见 GEM_OCR_MISSING_REASON 的注释。
            # 放在最前面：判据缺失是**配置问题**，不该等点开面板才发现
            ctx["fail_reason"] = GEM_OCR_MISSING_REASON
            ctx["zoom_verdict"] = "retry"
            logger.error("[宝石] manifest 里没有 `gem_dialog_title`（text_fields）"
                         "或 OCR 未启用，双确认缺 OCR 腿，本轮中止")
            return
        target = self._pending_gem
        if target is None:
            ctx["zoom_verdict"] = "retry"
            return
        self._click_xy(target[0], target[1], anchor="gem_icon")
        self._sleep(_ZOOM_SETTLE)
        if self._wait_for("gem_mine_zoom", _ZOOM_TIMEOUT):
            self._gem_fail_streak = 0
            ctx["zoom_verdict"] = "confirmed"
            logger.info("[宝石] 候选 (%.0f,%.0f) 放大后确认是宝石", *target)
            return
        self._give_up_candidate(ctx, target, "放大后认不出宝石", "zoom_verdict")

    # ---- ZOOM_CONFIRM → COLLECT_DIALOG ----

    def _open_collect_dialog(self, ctx):
        """在放大后的宝石画面上再点一下 → 采集框 → 双确认第二腿。

        `collect_dialog`（框在不在）+ `gem_dialog_title`（标题 OCR 读出
        「宝石」）。地图上「一小坨白色」既可能是宝石也可能是别的矿，只有
        图标腿会误派一支队伍去采 20 分钟。
        """
        ctx.pop("fail_reason", None)
        target = self._pending_gem
        if target is not None:
            self._click_xy(target[0], target[1], anchor="gem_mine_zoom")
        if self._wait_for("collect_dialog", _DIALOG_TIMEOUT):
            if self._find("gem_dialog_title") is not None:
                self._gem_fail_streak = 0
                ctx["dialog_verdict"] = "gem"
                logger.info("[宝石] 采集框 + 标题 OCR 双确认通过")
                return
            logger.info("[宝石] 采集框出来了但标题不是「宝石」（别的资源），跳过")
        else:
            logger.info("[宝石] 第二下没弹出采集框")
        self._give_up_candidate(ctx, target, "没拿到宝石采集框", "dialog_verdict")

    def _give_up_candidate(self, ctx, target, why: str, verdict_key: str):
        """放弃一个候选：**必须记进 `_tried_gems`**，否则同一颗宝石会被
        反复点开、反复关掉，整轮卡死在原地。

        连续放弃到 `_GEM_CONFIRM_MAX` 个就写 `gem_confirm_failed` 干净中止
        本轮（不抛异常）。
        """
        if target is not None:
            self._tried_gems.add(self._gem_key(target))
        self._gem_fail_streak += 1
        self._close_detail()
        ctx[verdict_key] = "retry"
        logger.warning("[宝石] 候选 %s %s（第 %s 次），跳过继续扫",
                       ("(%.0f,%.0f)" % target) if target else "（未知位置）",
                       why, self._gem_fail_streak)
        if self._gem_fail_streak >= _GEM_CONFIRM_MAX:
            ctx["fail_reason"] = GEM_CONFIRM_FAILED_REASON
            logger.warning("[宝石] 连续 %s 个候选都确认不了，本轮中止",
                           self._gem_fail_streak)

    def _close_detail(self):
        """关掉放大后的详情面板，回资源查看视角。

        用 `back_from_detail`（详情面板右上角的返回），失配时兜底点一次
        面板外的空地——**不能只靠点空地**：点在地图上的资源上会再开一个
        面板，越关越乱。
        """
        if self._click("back_from_detail"):
            self._sleep(_ZOOM_SETTLE)
            return
        self._click_xy(200, 900)
        self._sleep(_ZOOM_SETTLE)
```

```python
    两边靠字符串字面量对齐（SM 写 ctx["fail_reason"]，runner 按值查表），

    from rok_assistant.workers.runner import _ENV_FAIL_REASONS
    from rok_assistant.workers.state_machine import NORMALIZE_STUCK_REASON

             ("_NO_AP_REASON", "_AP_STUCK_REASON", "_STALE_TARGET_REASON")]
```

```python
    _sm_factory_for(...)             怎么给一个角色造状态机（必填）
    _pre_start(instances)            起 worker 之前的告警/校验
    _route_for(key, char)            要不要订阅 rally_launched
    _should_finish(runners)          「全都收工了」的判据
    _on_all_done(runners)            发 all_workers_done 之前的收尾
    _max_rounds()                    每 worker 的轮次上限
    _fills()                         统计条里的「填兵」列

共用的 `start` / `stop` / `stats` / `snapshot` 才是真正会漂的部分（上一轮
多模式框架评审 faebeeb 就翻在这里），所以它们**不拆**。
"""
from __future__ import annotations

from pathlib import Path
from typing import Callable

import cv2

from ..core.template_registry import TemplateRegistry
from ..infra.anti_detection import HumanProfile, JitteringHandleSource
from ..infra.app_paths import resolve_asset, user_dir
from ..infra.config import RootConfig
from ..infra.logger import get_logger
from ..workers.runner import WorkerRunner
from .event_bus import EventBus
from .preflight import preflight

logger = get_logger(__name__)

_STOP_TIMEOUT = 2.0   # GUI 主线程串行 join，超时要短，避免界面长时间冻结

# GUI 缩略图的编码上限。卡片上那块只有 298×150，原帧（1920×1080）按面积
# 算要多烧 16 倍的编码、传输与解码，而它每 2s 发一轮。
_THUMB_W, _THUMB_H = 480, 270


def _shrink_for_thumbnail(frame):
    """把一帧缩到缩略图尺寸（塞进 480×270 的框，保持宽高比，不拉伸）。

    只缩不放：分辨率没调对的模拟器给了更小的画面时，放大只会糊，而卡片
    本来就画得下。返回新数组，**不动原帧**——识别链还在读它，尺寸一变
    所有 ROI 坐标全错位。
    """
    h, w = frame.shape[:2]
    scale = min(_THUMB_W / w, _THUMB_H / h, 1.0)
    if scale >= 1.0:
        return frame
    # INTER_AREA 是缩小场景的正确插值（INTER_LINEAR 缩下来会有摩尔纹）
    return cv2.resize(frame, (max(1, round(w * scale)), max(1, round(h * scale))),
                      interpolation=cv2.INTER_AREA)


class BaseCoordinator:
    """从 RootConfig 组装运行时：每实例 1 个 HandleSource（防检测包装），
    每实例第 1 个角色 1 个 WorkerRunner。

    v1 限制：每实例只跑第 1 个角色（多角色切换属 v2）。
    """

    def __init__(self, config: RootConfig, event_bus: EventBus | None = None,
                 template_dir: Path | None = None):
        self._config = config
        self._bus = event_bus or EventBus()
        # 默认用配置里的模板目录；测试可注入临时目录。配置里写的是相对路径，
        # 按 resource_dir() 解开——冻结后它是 _internal，不是 CWD。
        self._template_dir = Path(template_dir) if template_dir \
            else resolve_asset(self._config.app.template_dir)
        self.runners: dict[str, WorkerRunner] = {}
        # runner_key -> rally_launched 路由处理器（stop 时统一退订）。
        # 宝石模式不用它，恒空表——留着是为了 stop/_rollback 只有一份实现。
        self._routes: dict[str, Callable] = {}
        # 自然收工的 worker 逐个上报；全部报过 → 通知 GUI 复位按钮。
        # 用户点 Stop 的路径不上报（runner 只在 stopped_reason 非 None 时发）。
        self._stopped: dict[str, str] = {}
        self._bus.subscribe("worker_finished", self._on_worker_finished)
        self._running = False

    # ---- 子类钩子 ----

    def _instances(self):
        """本模式跑哪些实例。**必填**。"""
        raise NotImplementedError

    def _sm_factory_for(self, char, handle, recognizers, profile):
        """给一个角色造状态机。**必填**（`sm_factory` 每轮重建时会再调）。"""
        raise NotImplementedError

    def _pre_start(self, instances) -> None:
        """起 worker 之前的告警/校验。默认什么都不做。"""

    def _route_for(self, key: str, char):
        """要不要给这个角色订阅 `rally_launched`。返回 handler 或 None。

        默认 None：只有寨子模式有「车头发起 → 成员填兵」这条跨实例链路，
        宝石模式一台机器一个号、各干各的。
        """
        return None

    def _should_finish(self, runners: dict) -> bool:
        """「全部收工」的判据。默认：所有 runner 都上报过。"""
        return bool(runners) and set(runners) <= set(self._stopped)

    def _on_all_done(self, runners: dict) -> None:
        """发 `all_workers_done` 之前的收尾。默认什么都不做。"""

    def _max_rounds(self) -> int:
        return self._config.app.max_rounds

    def _fills(self) -> int:
        """统计条里的「填兵」列。默认 0。"""
        return 0

    # ---- 生命周期 ----

    def start(self) -> None:
        if self._running:
            return
        # worker_finished 的订阅与「运行」同寿：先撤后挂，保证恰好一份
        # （__init__ 已挂一份；自然收工 / stop / 回滚 都会撤掉）。unsubscribe
        # 幂等，重复撤订无副作用。
        self._bus.unsubscribe("worker_finished", self._on_worker_finished)
        self._bus.subscribe("worker_finished", self._on_worker_finished)
        # 复用同一 coordinator（自然收工后没 stop 就再 start）时，runners 里
        # 留的是已死线程的引用、_stopped 已满——不清的话第二轮第一台 worker
        # 一收工就会满足「全部收工」提前触发，把运行中的 Stop 按钮禁掉。
        self.runners.clear()
        self._stopped.clear()
        instances = self._instances()
        self._pre_start(instances)
        # 先置位再组装：若中途失败（如识别器装配失败、第 2 台 worker 起不来），
        # 必须走回滚，否则半启动的 runner 成孤儿、下次 start() 会重复拉起
        # （连接失败已由下面的预检在 worker 诞生前拦下，不走这条回滚）
        self._running = True
        try:
            # 预检先行：在任何 worker 诞生之前确认每台模拟器都截得到
            # screen_width×screen_height 的画面。放在装配识别器之前——
            # 识别器要加载 ONNX，好几秒，连不上的时候不该白花（spec §7.3）。
            handles = preflight(instances, self._config.app)
            recognizers = self._recognizers(instances)
            for inst in instances:
                # 预检已经建好并验过帧，这里直接复用——见 preflight 的 docstring
                handle = handles[inst.id]
                profile = HumanProfile(self._config.app.anti_detection)
                handle = JitteringHandleSource(handle, profile)
                for char in inst.characters[:1]:
                    self._spawn(inst, char, handle, recognizers, profile)
        except Exception:
            logger.exception("运行时启动失败，回滚已创建的 worker")
            self._rollback()
            raise

    def _fill_names(self, instances) -> list[str]:
        """配置点名的填兵目标车头名。

        只有寨子模式的 CharacterConfig 有 `fill_target_leaders`；其他模式
        `getattr` 取不到就是空表——**子类不用覆写这个方法**。
        """
        return sorted({
            t.name
            for inst in instances
            for char in inst.characters
            for t in getattr(char, "fill_target_leaders", [])
        })

    def _recognizers(self, instances):
        """装配 `{template_id: Recognizer}`：manifest + 可选 YOLO + 可选 OCR。"""
        return TemplateRegistry.load(
            self._template_dir / "manifest.yaml").build_recognizers(
            yolo_model=(resolve_asset(self._config.app.yolo_model)
                        if self._config.app.yolo_model else None),
            ocr_fallback=self._config.app.ocr_name_fallback,
            fill_names=self._fill_names(instances))

    def _spawn(self, inst, char, handle, recognizers, profile) -> None:
        key = f"{inst.id}:{char.id}"
        runner = WorkerRunner(
            instance_id=inst.id, char_id=char.id, char_name=char.name,
            sm_factory=lambda: self._sm_factory_for(char, handle, recognizers,
                                                    profile),
            handle_source=handle, event_bus=self._bus,
            human=profile,
            # 显式给失败截图目录：冻结后 CWD 可能是任意位置，runner 的
            # 默认 Path("recordings") 会把截图写到用户找不到的地方。
            screenshot_dir=user_dir() / "recordings",
            max_rounds=self._max_rounds(),
            max_consecutive_failures=self._config.app.max_consecutive_failures)
        self.runners[key] = runner
        route = self._route_for(key, char)
        if route is not None:
            self._routes[key] = route
            self._bus.subscribe("rally_launched", route)
        runner.start()

    def _rollback(self) -> None:
        for key, runner in self.runners.items():
            runner.stop(timeout=_STOP_TIMEOUT)
            self._warn_if_alive(key, runner)
        self.runners.clear()
        for handler in self._routes.values():
            self._bus.unsubscribe("rally_launched", handler)
        self._routes.clear()
        self._bus.unsubscribe("worker_finished", self._on_worker_finished)
        self._stopped.clear()
        self._running = False

    def _on_worker_finished(self, payload: dict) -> None:
        """一台 worker 自然收工。满足 `_should_finish` → 收尾 → 发
        `all_workers_done`。

        只在 payload 里的 key 确实在 self.runners 里时才计入，否则
        「上一次运行遗留的迟到事件」会凑数提前复位按钮。

        本方法在发布事件的 worker 线程里被同步调用，多台 worker 近同时收工
        就会并发跑这里。**顶部拍一次 runners 快照**，后续只读快照：不这么做
        的话子类的收尾钩子会在另一线程 `runners.clear()` 时抛 RuntimeError，
        而异常被 EventBus 吞掉 → all_workers_done 永远发不出去（GUI 按钮
        不复位）。快照而非加锁：收尾钩子会 join 被停线程，若持锁等待正在
        同一处理函数里等锁的线程就死锁了。
        """
        runners = dict(self.runners)          # 快照：并发收工时另一线程可能 clear()
        key = f"{payload.get('instance_id')}:{payload.get('char_id')}"
        if key not in runners:
            return
        self._stopped[key] = payload.get("stopped_reason", "")
        if not self._should_finish(runners):
            return
        self._on_all_done(runners)
        logger.info("全部 worker 已收工：%s", self._stopped)
        self._bus.publish("all_workers_done", {"reasons": dict(self._stopped)})
        self._running = False
        # 收工即退订：订阅与运行同寿。否则旧 coordinator 仍挂在同一根 bus 上，
        # 下一次运行中途会被它凑满再发一次 all_workers_done（把 Stop 禁掉）。
        # EventBus.publish 遍历的是 handler 列表副本，循环内退订安全。
        self._bus.unsubscribe("worker_finished", self._on_worker_finished)
        # 清掉陈旧状态：自然收工后 runners 里是已死线程的引用、_stopped 已满，
        # 留着会让复用时下一轮第一次上报就提前凑满。
        self._stopped.clear()
        self.runners.clear()

    def stop(self) -> None:
        leftover: dict[str, WorkerRunner] = {}
        for key, runner in self.runners.items():
            runner.stop(timeout=_STOP_TIMEOUT)
            if self._warn_if_alive(key, runner):
                # 超时未停的守护线程不能丢引用：留在 runners 里，
                # 用户再点一次 Stop（或进程退出前）仍有句柄可控可查
                leftover[key] = runner
        for handler in self._routes.values():
            self._bus.unsubscribe("rally_launched", handler)
        self._routes.clear()
        self._bus.unsubscribe("worker_finished", self._on_worker_finished)
        self.runners = leftover
        self._stopped.clear()
        self._running = False

    @staticmethod
    def _warn_if_alive(key: str, runner) -> bool:
        """WorkerRunner 未暴露线程属性，这里读私有 _thread 做超时判断。"""
        thread = getattr(runner, "_thread", None)
        if thread is not None and thread.is_alive():
            logger.warning("worker %s 在 %.1fs 内未停止，守护线程仍在运行；"
                           "引用已保留，可再次 stop", key, _STOP_TIMEOUT)
            return True
        return False

    def stats(self) -> dict:
        """运行汇总，给 GUI 统计条用。

        只读、幂等：GUI 定时器在**主线程**调它，而 worker 线程同时在写这些
        计数器。顶部 `list()` 拍一次快照——worker 自然收工时
        `_on_worker_finished` 会 `runners.clear()`，直接迭代会抛
        RuntimeError: dictionary changed size during iteration（与 `snapshot()`
        同一个坑）。普通 int 属性的读写在 CPython 下不会撕裂，最坏是某个
        数字晚一拍。

        `target` 是「设置个数」：`_max_rounds()` × 在跑的角色数。没有 runner
        时返回 0 而不是 max_rounds——界面此时该显示 0/0，不是一份假目标。
        """
        runners = list(self.runners.values())
        return {
            "ok": sum(r.rounds_ok for r in runners),
            "failed": sum(r.rounds_failed for r in runners),
            "env": sum(r.rounds_env for r in runners),
            "rounds": sum(r.rounds_done for r in runners),
            "fills": self._fills(),
            "target": self._max_rounds() * len(runners),
        }

    def snapshot(self, char_id: str) -> bytes | None:
        """该角色最近一帧的 JPEG bytes（GUI 缩略图用），无则 None。

        遍历副本：本方法在 GUI 主线程被定时器调用，而 worker 线程自然收工时
        会走 _on_worker_finished 里的 self.runners.clear()。clear() 落在迭代
        中途会抛 RuntimeError: dictionary changed size during iteration。

        **编码前先缩到缩略图尺寸**：卡片上那块只有 298×150，按面积算原帧
        （1920×1080）要多烧 16 倍的编码、传输与解码，而它每 2s 就发一轮。
        缩的是送出去给人看的那份，识别链读的仍是 `runner.last_frame` 原帧。
        """
        for runner in list(self.runners.values()):
            if runner.char_id == char_id:
                img = runner.last_frame
                if img is None:
                    return None
                ok, buf = cv2.imencode(".jpg", _shrink_for_thumbnail(img))
                return bytes(buf) if ok else None
        return None
```

```python
# src/rok_assistant/coordination/gem_runtime.py
"""宝石模式的协调器。

与寨子模式的 RuntimeCoordinator 相比，这里**没有**：事件总线跨实例路由、
动作账本、集结事件登记簿、车头/成员分工、集结前置门槛。宝石模式一台
模拟器一个号、各干各的，队列与宝石都是本机的事实（spec §2）。

共用的骨架（预检 / 装配识别器 / 起 worker / 收工上报 / 停止 / 统计 /
缩略图）全在 `BaseCoordinator`。
"""
from __future__ import annotations

from ..infra.config import AppMode
from ..workers.factory import create_gem_state_machine
from .base_runtime import BaseCoordinator


class GemCoordinator(BaseCoordinator):
    """每实例 1 个 HandleSource（防检测包装），每实例第 1 个角色 1 个
    WorkerRunner，各自扫图 / 采集，互不通信。"""

    def _instances(self):
        return self._config.instances_for(AppMode.GEM)

    def _sm_factory_for(self, char, handle, recognizers, profile):
        gem = self._config.gem
        return create_gem_state_machine(
            char, handle, recognizers,
            queue_limit=gem.queue_limit, scan_rings=gem.scan_rings,
            scan_drag_px=gem.scan_drag_px, scan_drag_ms=gem.scan_drag_ms,
            idle_wait_min=gem.idle_wait_min, idle_wait_max=gem.idle_wait_max,
            human=profile)

    def _max_rounds(self) -> int:
        """`gem.max_rounds`，**不是** `app.max_rounds`：宝石模式跑得慢
        （一轮含 20~30 分钟等待），轮次上限是它自己的旋钮（spec §9）。"""
        return self._config.gem.max_rounds
```

### Task 5.1: 真机批量抓帧工具

**为什么不是「再抓 11 帧」**：11 帧里同一个图标、同一种背景，模型学到的只是
「这张图上的这个东西」。资源视角的地形底色随昼夜/地貌变，图标大小随缩放变，
必须**跨会话、跨模拟器、跨地图位置**抓。

**为什么抓帧脚本不自己进资源视角**：进视角要长按地图按钮再点环形菜单，
那一步的坐标正是阶段 0 待验项 2 在裁定的东西。抓帧工具**假定用户已经站在
资源视角**，只做「螺旋拖动 + 逐停抓帧」——和仓库既有的
`logs/_cap_raider/` 那套「停在目标界面 → 抓帧」一致。这样工具的成败不押在
一个还没验的坐标上。

**Files:**
- Create: `tools/capture_gem_view.py`
- Modify: `raw_imgs/README.md`（登记新场景）

**Interfaces:**
- Consumes: `coordination.preflight`、`workers.gem_sm._scan_vector` / `_scan_total_steps`（阶段 3）
- Produces: `raw_imgs/08_gem_view/*.png`

```python
# -*- coding: utf-8 -*-
"""站在资源视角，螺旋拖动并逐停抓帧 —— 宝石类的训练素材采集。

**前提**：模拟器已登录、已手动进入资源视角（长按地图按钮 → 环形菜单 → 资源）。
本工具**不做**进视角那几步：那几步的坐标由阶段 0 待验项 2 裁定，把工具押在
一个还没验的坐标上不划算。用户每次换一张地图/换一台机器就重跑一次。

**轨迹复用运行时的那一份**：`_scan_vector` / `_scan_total_steps` 直接从
`workers/gem_sm.py` import，不在这里抄第二份。抄一份的代价不是重复代码，
是**训练分布与推理分布悄悄分叉**——改了状态机的螺旋半径忘了改抓帧脚本，
模型就在一个它没见过的尺度上被调用。

注意 `_scan_vector(k)` 返回的是**单位方向**（`(1,0)` / `(0,1)` / `(-1,0)` /
`(0,-1)`），不是像素位移；乘 `--drag-px` 才是这一步该拖多远——状态机里也是
这么用的（`_drag(dx * self._scan_drag_px, dy * self._scan_drag_px)`）。

用法（仓库根目录）：
    # 先看一眼会拍多少张、路径长什么样（不连模拟器）
    .venv/Scripts/python.exe -X utf8 tools/capture_gem_view.py --dry-run

    # 真抓：默认 3 圈、每停一拍、写到 raw_imgs/08_gem_view/
    .venv/Scripts/python.exe -X utf8 tools/capture_gem_view.py --rings 3

    # 想多抓几圈就加大 --rings；换一台模拟器用 --instance
"""
from __future__ import annotations

import argparse
import sys
import time
from pathlib import Path

import cv2

ROOT = Path(__file__).resolve().parent.parent
sys.path.insert(0, str(ROOT / "src"))

from rok_assistant.coordination.preflight import preflight      # noqa: E402
from rok_assistant.infra.app_paths import resolve_asset          # noqa: E402
from rok_assistant.infra.config import AppMode, load_config      # noqa: E402
from rok_assistant.workers.gem_sm import (                       # noqa: E402
    _scan_total_steps, _scan_vector)

OUT_ROOT = ROOT / "raw_imgs" / "08_gem_view"


def _drag(handle, dx: int, dy: int, ms: int) -> None:
    """按世界方向 (dx, dy) 拖屏：手指要往**反方向**走。

    与 `gem_sm._drag` 同一约定（那边有完整理由）：手指右滑，世界左移。
    """
    cx, cy = 900, 520
    handle.swipe(cx, cy, cx - dx, cy - dy, duration_ms=ms)


def main() -> int:
    ap = argparse.ArgumentParser()
    ap.add_argument("--rings", type=int, default=3)
    ap.add_argument("--drag-px", type=int, default=800)
    ap.add_argument("--drag-ms", type=int, default=500)
    ap.add_argument("--settle", type=float, default=1.0,
                    help="每停一拍后等画面稳定（秒）")
    ap.add_argument("--instance", default="",
                    help="只抓这一台（默认抓 gem 段的全部）")
    ap.add_argument("--dry-run", action="store_true")
    args = ap.parse_args()

    steps = _scan_total_steps(args.rings)
    # k 从 1 起（不是 0）：`_scan_vector` 的 k 是「第几步」，状态机里也是
    # `_scan_index += 1` 之后才调它。从 0 起会让头两步都是「向右一格」。
    seq = list(range(1, steps + 1))
    if args.dry_run:
        print(f"rings={args.rings} → {len(seq)} 个停留点，每点 1 帧")
        for k in seq:
            ux, uy = _scan_vector(k)
            print(f"  #{k:>3}  单位方向=({ux:>2},{uy:>2})  "
                  f"位移=({ux * args.drag_px:>5},{uy * args.drag_px:>5})")
        print(f"输出目录：{OUT_ROOT}")
        return 0

    cfg = load_config(ROOT / "config.yaml")
    instances = [i for i in cfg.instances_for(AppMode.GEM)
                 if not args.instance or i.id == args.instance]
    if not instances:
        print("config.yaml 的 gem 段没有实例（或 --instance 没匹配上）")
        return 1
    handles = preflight(instances, cfg.app)
    OUT_ROOT.mkdir(parents=True, exist_ok=True)

    saved = 0
    for inst in instances:
        handle = handles[inst.id]
        print(f"[{inst.id}] 开始螺旋抓帧（请确认此刻在资源视角）")
        for k in seq:
            ux, uy = _scan_vector(k)
            if k > 1:                   # 第 1 拍是起点，不用先拖
                _drag(handle, ux * args.drag_px, uy * args.drag_px,
                      args.drag_ms)
            time.sleep(args.settle)
            frame = handle.capture()
            if frame is None:
                print(f"  #{k} 抓帧失败（模拟器掉线？），中止这台")
                break
            name = f"gem_{inst.id}_{time.strftime('%Y%m%d_%H%M%S')}_{k:03d}.png"
            cv2.imwrite(str(OUT_ROOT / name), frame)
            saved += 1
        print(f"[{inst.id}] 完成")
    print(f"\n共写 {saved} 帧 → {OUT_ROOT}")
    print("下一步：tools/ingest_raw_imgs.py --scene 08_gem_view --dry-run")
    return 0


if __name__ == "__main__":
    sys.exit(main())
```

```python
from pathlib import Path
```

```python
    encoding="utf-8").splitlines()

per_frame, zero, total = [], [], 0
for img in imgs:

    ids = [int(l.split()[0]) for l in lab.read_text(encoding="utf-8").splitlines()
           if l.strip()] if lab.exists() else []

print(f"帧数 {len(imgs)}  宝石框 {total}  含宝石的帧 "

print(f"0 宝石框的帧 {len(zero)}（这些是负样本，正常）")

print("\n框最多的 10 帧（可能是串位/重复标）：")
```
